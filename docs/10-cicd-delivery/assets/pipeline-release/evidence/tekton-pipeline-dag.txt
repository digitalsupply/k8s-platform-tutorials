## compassio-cicd
tasks:
  clone                    runAfter=-                                      when=-
  check-changes            runAfter=clone                                  when=-
  build-push-backend       runAfter=clone                                  when=$(tasks.check-changes.results.need-build) in ['true']
  build-push-frontend      runAfter=clone                                  when=$(tasks.check-changes.results.need-build) in ['true']
  promote-gitops           runAfter=build-push-backend,build-push-frontend when=$(tasks.check-changes.results.need-build) in ['true']
workspaces: ['shared-workspace']
